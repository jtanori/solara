// Server-only: keep hashing on the server. Do not import this module from a Client Component.
// Next.js `import "server-only"` would enforce this at build, but it breaks plain `tsx --test`
// imports, so we document the invariant here and rely on the build-time check.
// See P3 in PR #17 review.
import { randomBytes, timingSafeEqual } from "node:crypto";
import { ScryptBusyError, scryptPool } from "@/lib/scrypt-pool";

export async function hashPassword(password: string): Promise<string> {
  const salt = randomBytes(16).toString("hex");
  try {
    const hash = await scryptPool.run(password, salt);
    return `${salt}:${hash.toString("hex")}`;
  } catch (error) {
    if (error instanceof ScryptBusyError) throw error;
    throw new Error("Password hashing failed.");
  }
}

export async function verifyPassword(password: string, stored: string): Promise<boolean> {
  const [salt, hash] = stored.split(":");
  if (!salt || !hash) return false;
  // Explicit hex/length checks so failure is intentional, not an accident of Buffer truncation.
  if (!/^[0-9a-f]{32}$/.test(salt)) return false;
  if (!/^[0-9a-f]{128}$/.test(hash)) return false;
  const expected = Buffer.from(hash, "hex");
  let actual: Buffer;
  try {
    actual = await scryptPool.run(password, salt);
  } catch (error) {
    if (error instanceof ScryptBusyError) throw error;
    return false;
  }
  return expected.length === actual.length && timingSafeEqual(expected, actual);
}

import { scrypt } from "node:crypto";
import { promisify } from "node:util";
import { Semaphore, SemaphoreQueueFullError } from "@/lib/semaphore";

export type ScryptRunner = (
  password: string,
  salt: string,
  keylen: number,
) => Promise<Buffer>;

const defaultRunner = promisify(scrypt) as ScryptRunner;

export class ScryptBusyError extends Error {
  constructor(message = "Password hashing capacity is exhausted.") {
    super(message);
    this.name = "ScryptBusyError";
  }
}

export class ScryptPool {
  private readonly semaphore: Semaphore;

  constructor(
    private readonly runner: ScryptRunner = defaultRunner,
    maxConcurrent = 4,
    maxQueue = 256,
  ) {
    this.semaphore = new Semaphore(maxConcurrent, maxQueue);
  }

  get activeCount(): number {
    return this.semaphore.activeCount;
  }

  get queuedCount(): number {
    return this.semaphore.queuedCount;
  }

  async run(password: string, salt: string): Promise<Buffer> {
    try {
      await this.semaphore.acquire();
    } catch (error) {
      if (error instanceof SemaphoreQueueFullError) throw new ScryptBusyError();
      throw error;
    }

    try {
      return await this.runner(password, salt, 64);
    } finally {
      this.semaphore.release();
    }
  }
}

export const scryptPool = new ScryptPool();

export class SemaphoreQueueFullError extends Error {
  constructor(message = "Semaphore queue is full.") {
    super(message);
    this.name = "SemaphoreQueueFullError";
  }
}

export class Semaphore {
  private active = 0;
  private waiters: Array<() => void> = [];

  constructor(
    private readonly max: number,
    private readonly maxQueue: number,
  ) {
    if (max < 1) throw new Error("Semaphore max must be at least 1.");
    if (maxQueue < 0) throw new Error("Semaphore maxQueue must be non-negative.");
  }

  get activeCount(): number {
    return this.active;
  }

  get queuedCount(): number {
    return this.waiters.length;
  }

  tryAcquire(): boolean {
    if (this.active >= this.max) return false;
    this.active++;
    return true;
  }

  async acquire(): Promise<void> {
    if (this.active < this.max) {
      this.active++;
      return;
    }
    if (this.waiters.length >= this.maxQueue) {
      throw new SemaphoreQueueFullError();
    }
    await new Promise<void>((resolve) => {
      this.waiters.push(resolve);
    });
    // The permit is transferred by release(); active already counts this holder.
  }

  release(): void {
    const next = this.waiters.shift();
    if (next) {
      next();
      return;
    }
    if (this.active > 0) this.active--;
  }
}

async function handleSignup(
  request: Request,
  deps: AuthDeps,
  email: string,
  name: string,

  if (name.length < 1 || name.length > 100) {
    return {
      response: NextResponse.json(
        { error: es ? "Introduce tu nombre." : "Please enter your name." },

    AUTH_RATE_LIMITS.signupEmail,
    "signup_email",

    return { response: tooManyAttempts(es, emailLimit.retryAfterSeconds), tasks: [] };

  const appUrl = canonicalAppUrl(deps.appUrl);

  const [created] = await db.insert(users).values({
    name,
    email,
    passwordHash: PENDING_PASSWORD_HASH,
    role: "user",
    plan: "free",

    id: users.id,
    email: users.email,
    name: users.name,
    emailVerifiedAt: users.emailVerifiedAt,
  });

  if (created) {
    return {
      response: NextResponse.json({ ok: true }),
      tasks: [claimTask(created, deps.locale, mail, appUrl, context)],
    };
  }

  const [existing] = await db
    .select({
      id: users.id,
      email: users.email,
      name: users.name,
      emailVerifiedAt: users.emailVerifiedAt,
    })
    .from(users)
    .where(eq(users.email, email))
    .limit(1);

  if (!existing) {
    return { response: NextResponse.json({ ok: true }), tasks: [] };
  }

  return {
    response: NextResponse.json({ ok: true }),
    tasks: [existing.emailVerifiedAt
      ? accountExistsTask(existing, deps.locale, mail, appUrl, context)
      : claimTask(existing, deps.locale, mail, appUrl, context)],
  };
}

async function handleLogin(
  request: Request,
  deps: AuthDeps,
  email: string,
  password: string,
): Promise<AuthOutcome> {
  const es = deps.locale === "es";
  try {
    await loginAdmission.acquire();
  } catch (error) {
    if (error instanceof SemaphoreQueueFullError) {
      return { response: serviceUnavailable(es), tasks: [] };
    }
    throw error;
  }

  try {
    const ip = clientIp(request);
    const context = requestMeta(request);
    const ipFailureKey = keyFor("login:ip-failures", ip);
    const accountKey = keyFor("login:account", email);

    const outcome = await withLoginLocks(ip, email, async (tx) => {
      const ipLimit = await consumeRateLimitTx(
        tx,
        keyFor("login:ip", ip),
        AUTH_RATE_LIMITS.loginIp,
        "login_ip",
        context,
      );
      if (!ipLimit.allowed) {
        return { kind: "blocked" as const, retryAfter: ipLimit.retryAfterSeconds };
      }

      const ipFailureState = await getRateLimitStateTx(tx, ipFailureKey);
      const now = Date.now();
      if (
        ipFailureState
        && ipFailureState.expiresAt.getTime() > now
        && (
          (ipFailureState.blockedUntil?.getTime() ?? 0) > now
          || ipFailureState.count >= AUTH_RATE_LIMITS.loginIpFailures.limit
        )
      ) {
        const retryDate = ipFailureState.blockedUntil && ipFailureState.blockedUntil.getTime() > now
          ? ipFailureState.blockedUntil
          : ipFailureState.expiresAt;
        return { kind: "blocked" as const, retryAfter: retryAfterSeconds(retryDate) };
      }

      const accountState = await getRateLimitStateTx(tx, accountKey);
      if (accountState?.blockedUntil && accountState.blockedUntil.getTime() > now) {
        return { kind: "blocked" as const, retryAfter: retryAfterSeconds(accountState.blockedUntil) };
      }

      const [user] = await tx
        .select()
        .from(users)
        .where(eq(users.email, email))
        .limit(1);

      let passwordMatches: boolean;
      try {
        passwordMatches = await (deps.verify ?? verifyPassword)(
          password,
          passwordHashForLogin(user?.passwordHash),
        );
      } catch (error) {
        if (error instanceof ScryptBusyError) return { kind: "busy" as const };
        throw error;
      }

      if (!user || !passwordMatches || !user.emailVerifiedAt) {
        await recordAuthFailureTx(tx, accountKey, ACCOUNT_FAILURE_BACKOFF, context);
        const ipFailure = await consumeRateLimitTx(
          tx,
          ipFailureKey,
          AUTH_RATE_LIMITS.loginIpFailures,
          "login_ip_failures",
          context,
        );
        return {
          kind: "invalid" as const,
          ipRetryAfter: ipFailure.allowed ? null : ipFailure.retryAfterSeconds,
        };
      }

      await clearAuthFailuresTx(tx, accountKey);
      const session = await createSessionInTransaction(tx, user.id);
      return { kind: "ok" as const, user, session };
    });

    if (outcome.kind === "busy") {
      return { response: serviceUnavailable(es), tasks: [] };
    }
    if (outcome.kind === "blocked") {
      return { response: tooManyAttempts(es, outcome.retryAfter), tasks: [] };
    }
    if (outcome.kind === "invalid") {
      if (outcome.ipRetryAfter) {
        return { response: tooManyAttempts(es, outcome.ipRetryAfter), tasks: [] };
      }
      return {
        response: NextResponse.json(
          {
            error: es
              ? "El correo o la contraseña son incorrectos."
              : "Email or password is incorrect.",
          },
          { status: 401 },
        ),
        tasks: [],
      };
    }

    return {
      response: sessionResponse(outcome.user.name, outcome.session.token, outcome.session.expiresAt),
      tasks: [],
    };
  } finally {
    loginAdmission.release();
  }
}

export async function handleAuth(request: Request, deps: AuthDeps): Promise<AuthOutcome> {
  const es = deps.locale === "es";
  let data: {
    mode?: unknown;
    email?: unknown;
    password?: unknown;
    name?: unknown;
  };
  try {
    data = await request.json() as typeof data;
  } catch {
    return {
      response: NextResponse.json(
        { error: es ? "Solicitud no válida." : "Invalid request." },
        { status: 400 },
      ),
      tasks: [],
    };
  }

  const email = typeof data.email === "string" ? data.email.trim().toLowerCase() : "";
  const password = typeof data.password === "string" ? data.password : "";
  const name = typeof data.name === "string" ? data.name.trim() : "";

  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || email.length > 254) {
    return {
      response: NextResponse.json(
        { error: es ? "Introduce un correo válido." : "Enter a valid email." },
        { status: 400 },
      ),
      tasks: [],
    };
  }

  try {
    if (data.mode === "signup") {
      return await handleSignup(request, deps, email, name);
    }

    if (data.mode === "login") {
      if (password.length < 10 || password.length > 128) {
        return {
          response: NextResponse.json(
            {
              error: es
                ? "Introduce una contraseña de 10 a 128 caracteres."
                : "Enter a password of 10–128 characters.",
            },
            { status: 400 },
          ),
          tasks: [],
        };
      }
      return await handleLogin(request, deps, email, password);
    }

    return {
      response: NextResponse.json(
        { error: es ? "Solicitud no válida." : "Invalid request." },
        { status: 400 },
      ),
      tasks: [],
    };

async function consumeRateLimitIn(
  tx: DbTransaction,
  key: string,
  policy: RateLimitPolicy,
): Promise<RateLimitResult> {
  const now = new Date();

  await tx.insert(rateLimits).values({
    key,
    windowStart: now,
    count: 0,
    failureCount: 0,
    expiresAt: now,
  }).onConflictDoNothing();

  const [row] = await tx.select().from(rateLimits).where(eq(rateLimits.key, key)).for("update");
  if (!row) throw new Error("Rate limit row unavailable.");

  const expired = row.expiresAt.getTime() <= now.getTime();
  const count = expired ? 1 : row.count + 1;
  const failureCount = expired ? 0 : row.failureCount;
  const blockedUntil = expired ? null : row.blockedUntil;
  const expiresAt = expired
    ? new Date(now.getTime() + policy.windowSeconds * 1000)
    : row.expiresAt;
  const blocked = blockedUntil !== null && blockedUntil.getTime() > now.getTime();
  const overLimit = count > policy.limit;
  const triggered = blocked || overLimit;
  const nextBlockedUntil = blockedUntil ?? (overLimit ? expiresAt : null);

  await tx.update(rateLimits).set({
    windowStart: expired ? now : row.windowStart,
    count,
    failureCount,
    blockedUntil: nextBlockedUntil,
    expiresAt,
    lastTriggeredAt: triggered ? now : row.lastTriggeredAt,
  }).where(eq(rateLimits.key, key));

  const retryUntil = nextBlockedUntil ?? expiresAt;
  return {
    allowed: !triggered,
    remaining: Math.max(0, policy.limit - count),
    retryAfterSeconds: Math.max(1, Math.ceil((retryUntil.getTime() - now.getTime()) / 1000)),
    resetAt: expiresAt,
    blockedUntil: nextBlockedUntil,
  };
}

  kind: string,
  context: RateLimitContext = {},
): Promise<RateLimitResult> {

  return result;
}

export type MutationOriginDecision =
  | { allowed: true; reason: "same-origin" | "same-fetch-site" }
  | {
      allowed: false;
      reason: "missing-origin" | "invalid-origin" | "foreign-origin";
    };

export function evaluateMutationOrigin(input: {
  origin: string | null;
  secFetchSite: string | null;
  appUrl: string | undefined;
  requestUrl: string;
  trustedOrigins?: string[];
}): MutationOriginDecision {
  const trusted = new Set<string>();
  for (const value of [input.appUrl, input.requestUrl, ...(input.trustedOrigins ?? [])]) {
    const origin = normalizedOrigin(value);
    if (origin) trusted.add(origin);
  }

  if (!input.origin) {
    const secFetchSite = input.secFetchSite?.trim().toLowerCase();
    if (secFetchSite === "same-origin") {
      return { allowed: true, reason: "same-fetch-site" };
    }
    return { allowed: false, reason: "missing-origin" };
  }

  const origin = normalizedOrigin(input.origin);
  if (!origin) return { allowed: false, reason: "invalid-origin" };
  if (trusted.has(origin)) return { allowed: true, reason: "same-origin" };
  return { allowed: false, reason: "foreign-origin" };
}

  const pathname = request.nextUrl.pathname;
  const requestId = crypto.randomUUID();

  if (isGuardedApiPath(pathname) && isMutatingMethod(request.method)) {
    const decision = evaluateMutationOrigin({
      origin: request.headers.get("origin"),
      secFetchSite: request.headers.get("sec-fetch-site"),
      appUrl: process.env.APP_URL,
      requestUrl: request.url,

      const response = NextResponse.json({ error: "Invalid request origin." }, { status: 403 });
      response.headers.set("x-request-id", requestId);
      return response;
    }

    requestHeaders.set("x-request-id", requestId);
    const response = NextResponse.next({ request: { headers: requestHeaders } });
    response.headers.set("x-request-id", requestId);
    return response;

export async function withDeadline<T>(promise: Promise<T>, deadlineMs: number): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  const timeout = new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error("readiness_timeout")), deadlineMs);

  try {

  if (readinessPool) return readinessPool;

    max: 1,
    idleTimeoutMillis: 30_000,

  });

export async function probeReadiness(pool: Pool): Promise<ReadinessSnapshot> {
  let client: PoolClient;
  try {
    client = await pool.connect();
  } catch {
    return { dbOk: false, missingTables: [], missingColumns: [], missingIndexes: [], appliedHashes: [], authOk: false };
  }

  try {
    await client.query("begin");
    await client.query(`set local statement_timeout = '${READINESS_STATEMENT_TIMEOUT_MS}ms'`);
    await client.query("select 1");

    const tableRows = await client.query<{ name: string; reg: string | null }>(
      "select name, to_regclass(name) as reg from unnest($1::text[]) as name",
      [REQUIRED_TABLES],
    );
    const missingTables = tableRows.rows.filter(row => !row.reg).map(row => row.name);

    const columns = await client.query<ColumnRow>(`
      select table_name, column_name, is_nullable, data_type, column_default
      from information_schema.columns
      where table_schema = 'public'
        and table_name in ('users', 'rate_limits', 'email_tokens')
    `);
    const missingColumns = findMissingColumns(columns.rows);

    const indexes = await client.query<{ indexname: string }>(
      "select indexname from pg_indexes where schemaname = 'public' and indexname = any($1)",
      [[...REQUIRED_INDEXES]],
    );
    const presentIndexes = new Set(indexes.rows.map(row => row.indexname));
    const missingIndexes = REQUIRED_INDEXES.filter(index => !presentIndexes.has(index));

    const migrationTable = await client.query<{ reg: string | null }>(
      "select to_regclass('drizzle.__drizzle_migrations') as reg",
    );
    const appliedHashes = migrationTable.rows[0]?.reg
      ? (await client.query<{ hash: string }>("select hash from drizzle.__drizzle_migrations")).rows.map(row => row.hash)
      : [];

    const usersPresent = tableRows.rows.some(row => row.name === "public.users" && row.reg);
    let authOk = false;
    if (usersPresent) {
      try {
        await client.query("select 1 from public.users limit 1");
        authOk = true;
      } catch {
        authOk = false;
      }
    }

    await client.query("commit");
    return { dbOk: true, missingTables, missingColumns, missingIndexes, appliedHashes, authOk };
  } catch {
    await client.query("rollback").catch(() => {});
    return { dbOk: false, missingTables: [], missingColumns: [], missingIndexes: [], appliedHashes: [], authOk: false };
  } finally {
    client.release();
  }
}

export function buildReadiness(
  snapshot: ReadinessSnapshot,
  expected: MigrationManifestEntry[] = EXPECTED_MIGRATIONS,
): Readiness {
  const expectedHashes = new Set(expected.map(entry => entry.hash));
  const appliedHashes = new Set(snapshot.appliedHashes);
  const pending = expected.filter(entry => !appliedHashes.has(entry.hash)).length;
  const unexpected = snapshot.appliedHashes.filter(hash => !expectedHashes.has(hash)).length;
  const ok = snapshot.dbOk
    && snapshot.missingTables.length === 0
    && snapshot.missingColumns.length === 0
    && snapshot.missingIndexes.length === 0
    && pending === 0
    && unexpected === 0
    && snapshot.authOk;

  return {
    ok,
    db: snapshot.dbOk ? "ok" : "error",
    schema: {
      missing: snapshot.dbOk ? snapshot.missingTables : [],
      missingColumns: snapshot.dbOk ? snapshot.missingColumns : [],
      missingIndexes: snapshot.dbOk ? snapshot.missingIndexes : [],
    },
    migrations: { pending: snapshot.dbOk ? pending : expected.length, unexpected },
    auth: snapshot.dbOk && snapshot.authOk ? "ok" : "degraded",
  };
}

export async function evaluateReadiness(
  probe: () => Promise<ReadinessSnapshot>,
  expected: MigrationManifestEntry[] = EXPECTED_MIGRATIONS,
  deadlineMs: number = READINESS_DEADLINE_MS,
): Promise<Readiness> {
  return buildReadiness(await withDeadline(Promise.resolve().then(probe), deadlineMs), expected);
}

export async function getReadiness(): Promise<Readiness> {
  return evaluateReadiness(() => probeReadiness(getReadinessPool()));
}

/**
 * Resolve internal-document (usage-guide) access.
 *
 * Development: return all guides and honor a valid `?role=` override.
 * Non-development: deny unauthenticated access; signed-in users receive exactly one guide
 * matching their real role; ignore `?role=`.
 *
 * This is the testable core — no `server-only` guard. Import from
 * `@/lib/usage-guide-access` in server components to get the build-time
 * client-import protection.
 */
export function resolveUsageGuideAccess(
  options: ResolveUsageGuideAccessOptions,
): ResolvedUsageGuideAccess | null {
  const { environment, isDevelopment, user, requestedRole, locale = "en" } = options;
  const isDev = isDevelopment ?? isDevelopmentMode(environment);

  if (isDev) {
    const guides = usageGuides(locale);
    const activeAudience: UsageAudience = isUsageAudience(requestedRole)
      ? requestedRole
      : usageAudienceFor(user);
    return { guides, activeAudience };
  }

  // Non-development: deny anonymous
  if (!user) return null;

  const audience = usageAudienceFor(user);
  // usageGuides() is server-side static data; we load all 6 then filter to the single allowed
  // guide. The returned `guides` array is pre-filtered and must be passed through verbatim —
  // do not forward the full array to the client (see #13 "no other role content" criterion).
  const guides = usageGuides(locale);
  const single = guides.find((guide) => guide.id === audience);
  // Fail closed: unknown audience means deny (should not happen with valid Role mapping)
  if (!single) return null;
  // Hide internal-document section links (/setup, /blueprint) outside development.
  // These are not public in prod; they must not appear in HTML or RSC payload.
  const filteredSections = single.sections.map((section) => {
    if (section.link && isInternalDocumentLink(section.link.href)) {
      const { link: _ignored, ...rest } = section;
      return rest;
    }
    return section;
  });
  const filteredGuide: AudienceGuide = { ...single, sections: filteredSections };
  // Ignore requestedRole entirely outside development
  return { guides: [filteredGuide], activeAudience: audience };
}

  return href.startsWith("/setup") || href.startsWith("/blueprint");
}

import { NextResponse } from "next/server";
import { log, requestMeta } from "@/lib/log";
import {
  AUTH_RATE_LIMITS,
  clientIp,
  consumeRateLimit,
  keyFor,
} from "@/lib/rate-limit";

export const runtime = "nodejs";

const MAX_BODY_BYTES = 16_384;
const ALLOWED_TYPES = new Set([
  "application/csp-report",
  "application/reports+json",
  "application/json",
]);

function safePath(value: unknown): string | undefined {
  if (typeof value !== "string" || !value) return undefined;
  try {
    return new URL(value).pathname;
  } catch {
    return value.startsWith("/") ? value.split("?")[0] : undefined;
  }
}

function safeBlocked(value: unknown): string | undefined {
  if (typeof value !== "string" || !value) return undefined;
  if (value === "inline" || value === "eval" || value === "data" || value === "blob") {
    return value;
  }
  try {
    return new URL(value).origin;
  } catch {
    return undefined;
  }
}

function readReport(value: unknown): Record<string, unknown> | null {
  if (!value || typeof value !== "object") return null;
  if ("csp-report" in value) {
    const report = (value as { "csp-report"?: unknown })["csp-report"];
    return report && typeof report === "object" ? report as Record<string, unknown> : null;
  }
  if ("body" in value) {
    const body = (value as { body?: unknown }).body;
    return body && typeof body === "object" ? body as Record<string, unknown> : null;
  }
  return value as Record<string, unknown>;
}

export async function POST(request: Request) {
  const context = requestMeta(request);
  const limit = await consumeRateLimit(
    keyFor("csp:ip", clientIp(request)),
    AUTH_RATE_LIMITS.cspReportIp,
    "csp_report_ip",
    context,
  );
  if (!limit.allowed) return new NextResponse(null, { status: 204 });

  const contentType = request.headers.get("content-type")?.split(";")[0].trim().toLowerCase();
  if (!contentType || !ALLOWED_TYPES.has(contentType)) {
    return new NextResponse(null, { status: 415 });
  }

  const body = await request.text();
  if (body.length > MAX_BODY_BYTES) {
    return new NextResponse(null, { status: 413 });
  }

  let parsed: unknown;
  try {
    parsed = JSON.parse(body);
  } catch {
    return new NextResponse(null, { status: 400 });
  }

  const report = readReport(parsed);
  if (!report) {
    return new NextResponse(null, { status: 400 });
  }

  log.warn("csp.violation", {
    ...context,
    documentPath: safePath(report["document-uri"]),
    directive: report["violated-directive"] ?? report["effective-directive"],
    blocked: safeBlocked(report["blocked-uri"]),
    disposition: report.disposition,
  });

  return new NextResponse(null, { status: 204 });
}
