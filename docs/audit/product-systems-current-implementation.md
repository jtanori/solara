import Stripe from "stripe";

export type BillingCurrency = "mxn" | "usd";
export type PriceOption = { currency: BillingCurrency; amount: number; interval: string; intervalCount: number };

export function getStripe() {
  const key = process.env.STRIPE_SECRET_KEY;
  return key ? new Stripe(key) : null;
}
export function priceIdFor(currency: BillingCurrency) {
  return currency === "mxn" ? process.env.STRIPE_PRICE_ID_MXN : process.env.STRIPE_PRICE_ID_USD;
}
export function billingIsConfigured() {
  return !!(process.env.STRIPE_SECRET_KEY && process.env.STRIPE_WEBHOOK_SECRET && process.env.STRIPE_PRICE_ID_MXN);
}
export function isPremiumPrice(priceId: string) {
  const allowed = [process.env.STRIPE_PRICE_ID_MXN, process.env.STRIPE_PRICE_ID_USD, ...(process.env.STRIPE_PREMIUM_PRICE_IDS ?? "").split(",")]
    .map(value => value?.trim()).filter(Boolean);
  return allowed.includes(priceId);
}
export function appOrigin(request: Request) {
  const configured = process.env.APP_URL;
  if (configured) return new URL(configured).origin;
  const origin = request.headers.get("origin");
  return origin ? new URL(origin).origin : new URL(request.url).origin;
}
export async function getPriceCatalog(): Promise<PriceOption[]> {
  const stripe = getStripe();
  if (!stripe || !billingIsConfigured()) return [];
  const currencies: BillingCurrency[] = ["mxn", "usd"];
  const prices = await Promise.all(currencies.map(async currency => {
    const id = priceIdFor(currency);
    if (!id) return null;
    try {
      const price = await stripe.prices.retrieve(id);
      if (!price.active || price.currency !== currency || !price.recurring || price.unit_amount == null) return null;
      return { currency, amount: price.unit_amount, interval: price.recurring.interval, intervalCount: price.recurring.interval_count } satisfies PriceOption;
    } catch { return null; }
  }));
  return prices.filter((price): price is PriceOption => price !== null);
}

export function verifyStripeEvent(payload:string, signature:string, secret:string) {
  return Stripe.webhooks.constructEvent(payload,signature,secret);
}
export function subscriptionIdFrom(event:Stripe.Event):string|null {
  if(event.type==="checkout.session.completed"){
    const subscription=(event.data.object as Stripe.Checkout.Session).subscription;
    return typeof subscription==="string"?subscription:subscription?.id??null;
  }
  if(event.type==="customer.subscription.created"||event.type==="customer.subscription.updated"||event.type==="customer.subscription.deleted")return (event.data.object as Stripe.Subscription).id;
  if(event.type==="invoice.paid"||event.type==="invoice.payment_failed"||event.type==="invoice.payment_succeeded"){
    const invoice=event.data.object as Stripe.Invoice;
    const legacy=(invoice as unknown as {subscription?:string|{id:string}|null}).subscription;
    const subscription=invoice.parent?.subscription_details?.subscription??legacy;
    return typeof subscription==="string"?subscription:subscription?.id??null;
  }
  return null;
}

import type Stripe from "stripe";
import { eq, sql } from "drizzle-orm";
import { db } from "@/db";
import { stripeEvents, subscriptions, users } from "@/db/schema";
import { subscriptionIdFrom } from "@/lib/stripe-events";
import { isPremiumPrice } from "@/lib/billing";

export async function processStripeEvent(event:Stripe.Event,stripe:Stripe) {
  const id=subscriptionIdFrom(event);
  await db.transaction(async tx=>{
    const [claimed]=await tx.insert(stripeEvents).values({id:event.id,type:event.type}).onConflictDoNothing().returning({id:stripeEvents.id});
    if(!claimed||!id)return;
    // Stripe does not guarantee event order: serialize per subscription and fetch its latest state.
    await tx.execute(sql`select pg_advisory_xact_lock(hashtext(${id}))`);
    const subscription=await stripe.subscriptions.retrieve(id);
    const customerId=typeof subscription.customer==="string"?subscription.customer:subscription.customer.id;
    const metadataId=subscription.metadata.solara_user_id;
    let owner=metadataId&&/^[0-9a-f-]{36}$/i.test(metadataId)?(await tx.select({id:users.id,customerId:users.stripeCustomerId}).from(users).where(eq(users.id,metadataId)).limit(1))[0]:undefined;
    if(!owner)owner=(await tx.select({id:users.id,customerId:users.stripeCustomerId}).from(users).where(eq(users.stripeCustomerId,customerId)).limit(1))[0];
    if(!owner)return;
    if(owner.customerId&&owner.customerId!==customerId)throw new Error("Stripe customer mismatch");
    if(!owner.customerId)await tx.update(users).set({stripeCustomerId:customerId}).where(eq(users.id,owner.id));
    const item=subscription.items.data[0];if(!item)throw new Error("Subscription has no price item");
    const values={stripeSubscriptionId:subscription.id,userId:owner.id,stripeCustomerId:customerId,stripePriceId:item.price.id,status:subscription.status,currency:item.price.currency,grantsAccess:subscription.metadata.solara_product==="premium" && isPremiumPrice(item.price.id),currentPeriodEnd:item.current_period_end?new Date(item.current_period_end*1000):null,cancelAtPeriodEnd:subscription.cancel_at_period_end,updatedAt:new Date()};
    await tx.insert(subscriptions).values(values).onConflictDoUpdate({target:subscriptions.stripeSubscriptionId,set:values});
  });
}
